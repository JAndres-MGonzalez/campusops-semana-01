## Notas de ejecucion (desviaciones respecto a la guia)

- Secuencia real del backend: la guia usa resolve con baseVersion 1 y close con baseVersion 2. El servidor respondio 409 (transition_conflict y version_conflict, currentVersion 1) porque la incidencia estaba en estado assigned y hacia falta start antes. Salidas literales del primer intento:
~~~
{"code":"transition_conflict"}
HTTP_STATUS:409
{"code":"version_conflict","currentVersion":1}
HTTP_STATUS:409
~~~
- Secuencia valida usada despues: start (baseVersion 1), resolve (baseVersion 2), close (baseVersion 3). Ver las salidas de resolve y close mas arriba.
- Los cuerpos JSON de los curl se enviaron desde archivo (--data-binary @archivo) por el escape de comillas de PowerShell 5.1.
- La prueba temporal incluye un caso adicional propio: login/refresh exitoso y luego logout debe dejar persistedToken en null. Con el estado inicial, los casos de logout y refreshFailed no distinguen si el reset existe, asi que este caso permite detectar la falla controlada.
- Falla controlada: se comento la linea persistedToken = null del logout en src/infrastructure/session/auth-reducer.ts. Resultado: Tests: 1 failed, 6 passed, 7 total, expect(received).toBeNull() con Received: "t1". Revertida con git checkout y ningun cambio de codigo en el PR.

import { coordinateRefresh, redactForTelemetry } from './index';
import { createSessionManager } from '../infrastructure/session/manager';
import { createMemorySessionStorage } from '../infrastructure/session/memory-storage';

describe('week-06 session manual', () => {
  test('nominal: 401 concurrentes comparten una sola renovacion', () => {
    const summary = coordinateRefresh([
      { type: 'request401', requestId: 'a', generation: 0 },
      { type: 'request401', requestId: 'b', generation: 0 },
      { type: 'refreshSucceeded', generation: 1, token: 'course-token-1' },
    ]);
    expect(summary).toEqual({
      status: 'authenticated',
      activeGeneration: 1,
      refreshCalls: 1,
      retriedRequestIds: ['a', 'b'],
      persistedToken: 'course-token-1',
    });
  });

  test('boundary: una segunda generacion de 401 no dispara otra renovacion', () => {
    const summary = coordinateRefresh([
      { type: 'request401', requestId: 'a', generation: 0 },
      { type: 'refreshSucceeded', generation: 1, token: 't1' },
      { type: 'request401', requestId: 'b', generation: 1 },
    ]);
    expect(summary.refreshCalls).toBe(1);
    expect(summary.retriedRequestIds).toEqual(['a', 'b']);
  });

  test('failure: refreshFailed vuelve a no autenticado y olvida el token', () => {
    const summary = coordinateRefresh([
      { type: 'request401', requestId: 'a', generation: 0 },
      { type: 'refreshFailed', generation: 0 },
    ]);
    expect(summary.status).toBe('anonymous');
    expect(summary.persistedToken).toBeNull();
    expect(summary.refreshCalls).toBe(1);
  });

  test('failure: logout borra el estado persistido', () => {
    const summary = coordinateRefresh([{ type: 'logout' }]);
    expect(summary.persistedToken).toBeNull();
    expect(summary.status).toBe('anonymous');
  });

  test('boundary: expiracion con reloj controlable', async () => {
    let clock = 1_000;
    const storage = createMemorySessionStorage();
    const manager = createSessionManager({
      storage,
      now: () => clock,
      baseUrl: 'http://127.0.0.1:4310',
      fetchImpl: async (input: RequestInfo | URL) => {
        const url = String(input);
        if (url.includes('/v1/session/refresh')) {
          return new Response('{}', { status: 401 });
        }
        return new Response(
          JSON.stringify({ token: 't1', generation: 1, expiresInMs: 60_000 }),
          { status: 200, headers: { 'Content-Type': 'application/json' } },
        );
      },
    });
    await manager.login('technician-1');
    clock += 120_000; // el token ya expiro, sin sleep real
    expect(await manager.getValidToken()).toBeNull();
  });

  test('boundary: los tokens no quedan en claro para telemetria', () => {
    const redacted = redactForTelemetry({ token: 'course-valid-token', authorization: 'Bearer x' });
    expect(JSON.stringify(redacted)).not.toContain('course-valid-token');
    expect(JSON.stringify(redacted)).toContain('[REDACTED]');
  });
});
